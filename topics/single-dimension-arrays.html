9. Enter an array from user and find the k-th smallest value?

#include <stdio.h>
int main(void) {
    int a[100], n, k, temp;
    scanf("%d", &n);
    if (n < 1 || n > 100) return 1;

    for (int i = 0; i < n; i++) scanf("%d", &a[i]);
    printf("Enter k: ");
    scanf("%d", &k);
    if (k < 1 || k > n) {
        printf("Invalid k");
        return 1;
    }

    /* Sort the array in ascending order. */
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - 1 - i; j++) {
            if (a[j] > a[j + 1]) {
                temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("%d-th smallest value = %d", k, a[k - 1]);
    return 0;
}

    printf("Duplicate values: ");
    for (int i = 0; i < n; i++) {
        found = 0;
        for (int j = 0; j < i; j++)
            if (a[i] == a[j]) found = 1;

        if (!found) {

                    printf("%d ", a[i]);
                    break;
                }
            }
        }
    }

    return 0;
}

        alreadyCounted = 0;
        for (int j = 0; j < i; j++)
            if (a[i] == a[j]) alreadyCounted = 1;

            count = 1;

            if (count > 1) printf("%d occurs %d times\\n", a[i], count);
        }
    }

    return 0;
}

14. Enter an array from user and print all possible combinations of the elements?

#include <stdio.h>
void printSubsets(int a[], int n, int index, int chosen[], int count) {
    if (index == n) {
        if (count > 0) {
            printf("{ ");
            for (int i = 0; i < count; i++) printf("%d ", chosen[i]);
            printf("}\\n");
        }
        return;
    }

    /* Exclude the current element. */
    printSubsets(a, n, index + 1, chosen, count);

    /* Include the current element. */
    chosen[count] = a[index];
    printSubsets(a, n, index + 1, chosen, count + 1);
}

int main(void) {
    int a[20], chosen[20], n;
    scanf("%d", &n);
    if (n < 1 || n > 20) return 1;

    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    printf("All non-empty subsets (2^n - 1 combinations):\\n");
    printSubsets(a, n, 0, chosen, 0);
    return 0;
}

15. Enter an array from user and print all the palindrome numbers?

#include <stdio.h>
int isPalindrome(int number) {
    int original = number, reverse = 0;
    if (number < 0) return 0;

    do {
        reverse = reverse * 10 + number % 10;
        number /= 10;
    } while (number > 0);

    return original == reverse;
}

int main(void) {
    int a[100], n;
    scanf("%d", &n);
    if (n < 1 || n > 100) return 1;

    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    printf("Palindrome numbers: ");
    for (int i = 0; i < n; i++)
        if (isPalindrome(a[i])) printf("%d ", a[i]);

    return 0;
}

17. Merge the two sorted arrays into one array but in sorted order?

#include <stdio.h>
int main(void) {
    int a[100], b[100], c[200], n, m;
    int i = 0, j = 0, k = 0;

    scanf("%d", &n);
    if (n < 1 || n > 100) return 1;
    for (int x = 0; x < n; x++) scanf("%d", &a[x]);

    scanf("%d", &m);
    if (m < 1 || m > 100) return 1;
    for (int x = 0; x < m; x++) scanf("%d", &b[x]);

    /* Both input arrays must already be sorted in ascending order. */
    while (i < n && j < m) {
        if (a[i] <= b[j]) c[k++] = a[i++];
        else c[k++] = b[j++];
    }
    while (i < n) c[k++] = a[i++];
    while (j < m) c[k++] = b[j++];

    printf("Merged sorted array: ");
    for (int x = 0; x < k; x++) printf("%d ", c[x]);

    return 0;
}

18. Find the local minima in the array (an element is smaller than the element before and after it).

#include <stdio.h>
int main(void) {
    int a[100], n, found = 0;
    scanf("%d", &n);
    if (n < 3 || n > 100) {
        printf("Enter between 3 and 100 elements");
        return 1;
    }

    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    printf("Local minima: ");
    /* Only interior elements have both a previous and next neighbour. */
    for (int i = 1; i < n - 1; i++) {
        if (a[i] < a[i - 1] && a[i] < a[i + 1]) {
            printf("%d ", a[i]);
            found = 1;
        }
    }
    if (!found) printf("None");

    return 0;
}

20. Trapping Rain Water: Given non-negative bar heights of width 1, calculate how much water can be trapped.

#include <stdio.h>
int main(void) {
    int height[100], n;
    int left = 0, right, leftMax = 0, rightMax = 0;
    long long water = 0;

    scanf("%d", &n);
    if (n < 1 || n > 100) return 1;

    for (int i = 0; i < n; i++) {
        scanf("%d", &height[i]);
        if (height[i] < 0) {
            printf("Heights must be non-negative");
            return 1;
        }
    }

    right = n - 1;
    while (left < right) {
        if (height[left] <= height[right]) {
            if (height[left] >= leftMax) leftMax = height[left];
            else water += leftMax - height[left];
            left++;
        } else {
            if (height[right] >= rightMax) rightMax = height[right];
            else water += rightMax - height[right];
            right--;
        }
    }

    printf("Trapped water = %lld", water);
    return 0;
}
