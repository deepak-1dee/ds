            s[i] = (char)(first ? toupper((unsigned char)s[i]) : tolower((unsigned char)s[i]));

        }

}

20. Enter text and count characters, words, spaces, and lines.

#include <stdio.h>
#include <ctype.h>
int main() {
    char s[2000]; int i, chars = 0, words = 0, spaces = 0, lines = 0, inWord = 0;
    printf("Enter text (finish with Ctrl+D on Linux/Mac or Ctrl+Z then Enter on Windows):\n");
    while (fgets(s, sizeof(s), stdin) != NULL) {
        int hasContent = 0;
        for (i = 0; s[i] != '\0'; i++) {
            if (s[i] == '\n') { lines++; continue; }
            chars++;
            if (s[i] == ' ') spaces++;
            if (isspace((unsigned char)s[i])) inWord = 0;
            else { hasContent = 1; if (!inWord) words++; inWord = 1; }
        }
        if (hasContent && (i == 0 || s[i - 1] != '\n')) lines++;
    }
    printf("Characters (excluding newline) = %d\nWords = %d\nSpaces = %d\nLines = %d\n", chars, words, spaces, lines);
    return 0;
}

21. Print all permutations of the string ABCDE.

#include <stdio.h>
#include <string.h>
void permute(char s[], int left, int right) {
    int i; char temp;
    if (left == right) { printf("%s\n", s); return; }
    for (i = left; i <= right; i++) {
        temp = s[left]; s[left] = s[i]; s[i] = temp;
        permute(s, left + 1, right);
        temp = s[left]; s[left] = s[i]; s[i] = temp;
    }
}
int main() {
    char s[] = "ABCDE";
    permute(s, 0, (int)strlen(s) - 1);
    return 0;
}

24. Enter a sentence, a word to find, and a replacement word; replace whole-word matches.

#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main() {
    char sentence[1000], word[100], replacement[100], result[2000];
    int i = 0, j = 0, n, r, count = 0;
    printf("Enter a sentence: "); fgets(sentence, sizeof(sentence), stdin);
    sentence[strcspn(sentence, "\n")] = '\0';
    printf("Word to replace: "); scanf("%99s", word);
    printf("Replacement word: "); scanf("%99s", replacement);
    n = (int)strlen(word); r = (int)strlen(replacement);
    while (sentence[i] != '\0' && j < (int)sizeof(result) - 1) {
        if (strncmp(&sentence[i], word, n) == 0 &&
            (i == 0 || !isalnum((unsigned char)sentence[i - 1])) &&
            !isalnum((unsigned char)sentence[i + n])) {
            if (j + r >= (int)sizeof(result) - 1) break;
            memcpy(&result[j], replacement, r); j += r; i += n; count++;
        } else result[j++] = sentence[i++];
    }
    result[j] = '\0';
    printf("Updated sentence: %s\nReplacements: %d\n", result, count);
    return 0;
}

27. Display a string with user-selected alignment: 1 left, 2 center, 3 right (width 60).

#include <stdio.h>
#include <string.h>
int main() {
    char s[500]; int choice, width = 60, len, i, pad;
    printf("Enter text: "); fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0'; len = (int)strlen(s);
    printf("1. Left  2. Center  3. Right\nChoose alignment: "); scanf("%d", &choice);
    if (len > width) { printf("%s\n", s); return 0; }
    if (choice == 1) pad = 0;
    else if (choice == 2) pad = (width - len) / 2;
    else if (choice == 3) pad = width - len;
    else { printf("Invalid option.\n"); return 0; }
    for (i = 0; i < pad; i++) putchar(' ');
    printf("%s\n", s);
    return 0;
}

28. Create a small dictionary of 10 words with create, search, and delete operations.

#include <stdio.h>
#include <string.h>
#define SIZE 10
int main() {
    char words[SIZE][30], meanings[SIZE][100], w[30]; int count = 0, choice, i, found;
    while (1) {
        printf("\n1.Create  2.Search  3.Delete  4.Display  5.Exit\nChoice: ");
        if (scanf("%d", &choice) != 1) break;
        if (choice == 1) {
            if (count == SIZE) { printf("Dictionary is full.\n"); continue; }
            printf("Word: "); scanf("%29s", words[count]);
            printf("Meaning (one line, use underscores for spaces): "); scanf("%99s", meanings[count]);
            count++; printf("Word added.\n");
        } else if (choice == 2 || choice == 3) {
            printf("Enter word: "); scanf("%29s", w); found = -1;
            for (i = 0; i < count; i++) if (strcmp(words[i], w) == 0) { found = i; break; }
            if (found == -1) printf("Word not found.\n");
            else if (choice == 2) printf("%s: %s\n", words[found], meanings[found]);
            else {
                for (i = found; i < count - 1; i++) {
                    strcpy(words[i], words[i + 1]); strcpy(meanings[i], meanings[i + 1]);
                }
                count--; printf("Word deleted.\n");
            }
        } else if (choice == 4) {
            for (i = 0; i < count; i++) printf("%s: %s\n", words[i], meanings[i]);
        } else if (choice == 5) break;
        else printf("Invalid choice.\n");
    }
    return 0;
}
