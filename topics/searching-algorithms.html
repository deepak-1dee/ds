5. Enter an Integer Array from the User, Enter an Integer Value and Perform Binary Search Algorithm Through Iteration

#include <stdio.h>

int main() {
    int a[100], n, key, low, high, mid, found = -1;

    printf("Enter array size: ");
    scanf("%d", &n);

    if (n <= 0 || n > 100) {
        printf("Invalid array size\n");
        return 0;
    }

    printf("Enter array elements:\n");
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    // Sort array using bubble sort
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (a[j] > a[j + 1]) {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Sorted array: ");
    for (int i = 0; i < n; i++)
        printf("%d ", a[i]);

    printf("\nEnter value to search: ");
    scanf("%d", &key);

    low = 0;
    high = n - 1;

    while (low <= high) {
        mid = low + (high - low) / 2;

        if (a[mid] == key) {
            found = mid;
            break;
        }
        else if (key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    if (found != -1)
        printf("Element found at sorted index %d\n", found);
    else
        printf("Element not found\n");

    return 0;
}

#include <stdio.h>
#include <string.h>

int binarySearch(char str[], int low, int high, char key) {
    if (low > high)
        return -1;

    int mid = low + (high - low) / 2;

    if (str[mid] == key)
        return mid;

    if (key < str[mid])
        return binarySearch(str, low, mid - 1, key);

    return binarySearch(str, mid + 1, high, key);
}

int main() {
    char str[100], key;
    int n, result;

    printf("Enter a string: ");
    scanf("%99s", str);

    n = strlen(str);

    // Sort characters using bubble sort
    for (int i = 0; i < n - 1; i++) {

            if (str[j] > str[j + 1]) {
                char temp = str[j];
                str[j] = str[j + 1];
                str[j + 1] = temp;
            }
        }
    }

    printf("Sorted string: %s\n", str);

    printf("Enter character to search: ");
    scanf(" %c", &key);

    if (result != -1)
        printf("Character found at sorted index %d\n", result);
    else
        printf("Character not found\n");

    return 0;
}

9. Calculate the Best and Worst Complexity of the Linear Searching Algorithm with Exact Time Taken by the Algorithm on Integer Array

#include <stdio.h>
#include <time.h>
#include <limits.h>

int linearSearch(int a[], int n, int key, int *comparisons) {
    for (int i = 0; i < n; i++) {
        (*comparisons)++;

        if (a[i] == key)
            return i;
    }

    return -1;
}

int main() {
    int a[100], n, bestKey, worstKey;
    int bestComparisons = 0, worstComparisons = 0;
    int bestResult, worstResult;
    clock_t start, end;
    double bestTime, worstTime;

    printf("Enter array size: ");
    scanf("%d", &n);

    if (n <= 0 || n > 100) {
        printf("Invalid array size\n");
        return 0;
    }

    printf("Enter array elements:\n");
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    // Best case: key is the first element
    bestKey = a[0];

    start = clock();
    bestResult = linearSearch(a, n, bestKey, &bestComparisons);
    end = clock();

    bestTime = (double)(end - start) / CLOCKS_PER_SEC;

    // Find a value that is not present for the worst case
    worstKey = INT_MIN;

    while (1) {
        int exists = 0;

        for (int i = 0; i < n; i++) {
            if (a[i] == worstKey) {
                exists = 1;
                break;
            }
        }

        if (!exists)
            break;

        worstKey++;
    }

    start = clock();
    worstResult = linearSearch(a, n, worstKey, &worstComparisons);
    end = clock();

    worstTime = (double)(end - start) / CLOCKS_PER_SEC;

    printf("\nBest Case:\n");
    printf("Search key: %d\n", bestKey);
    printf("Result index: %d\n", bestResult);
    printf("Comparisons: %d\n", bestComparisons);
    printf("Time taken: %.9f seconds\n", bestTime);
    printf("Time complexity: O(1)\n");

    printf("\nWorst Case:\n");
    printf("Search key: %d (not present)\n", worstKey);
    printf("Result index: %d\n", worstResult);
    printf("Comparisons: %d\n", worstComparisons);
    printf("Time taken: %.9f seconds\n", worstTime);
    printf("Time complexity: O(n)\n");

    return 0;
}

10. Calculate the Best and Worst Complexity of the Binary Searching Algorithm with Exact Time Taken by the Algorithm on Integer Array

#include <stdio.h>
#include <time.h>
#include <limits.h>

int binarySearch(int a[], int n, int key, int *comparisons) {
    int low = 0, high = n - 1;

    while (low <= high) {
        int mid = low + (high - low) / 2;
        (*comparisons)++;

        if (a[mid] == key)
            return mid;

        if (key < a[mid])
            high = mid - 1;
        else
            low = mid + 1;
    }

    return -1;
}

    int bestComparisons = 0, worstComparisons = 0;
    int bestResult, worstResult;

        return 0;

        scanf("%d", &a[i]);

    for (int i = 0; i < n - 1; i++) {

            if (a[j] > a[j + 1]) {
                int temp = a[j];
                a[j] = a[j + 1];
                a[j + 1] = temp;
            }
        }
    }

    printf("Sorted array: ");
    for (int i = 0; i < n; i++)

    // Best case: middle element is found immediately
