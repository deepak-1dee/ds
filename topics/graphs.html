1. Enter the Vertices and Edges and Represent in Adjacency Matrix and Print It

#include <stdio.h>

int main() {
    int graph[50][50] = {0};
    int vertices, edges, u, v;

    printf("Enter number of vertices: ");
    scanf("%d", &vertices);

    if (vertices <= 0 || vertices > 50) {
        printf("Invalid number of vertices\n");
        return 0;
    }

    printf("Enter number of edges: ");
    scanf("%d", &edges);

    if (edges < 0) {
        printf("Invalid number of edges\n");
        return 0;
    }

    printf("Enter edges (u v), vertices numbered 0 to %d:\n",
           vertices - 1);

    for (int i = 0; i < edges; i++) {
        scanf("%d %d", &u, &v);

        if (u < 0 || u >= vertices || v < 0 || v >= vertices) {
            printf("Invalid edge. Try again.\n");
            i--;
            continue;
        }

        graph[u][v] = 1;
        graph[v][u] = 1;
    }

    printf("\nAdjacency Matrix:\n");

    for (int i = 0; i < vertices; i++) {
        for (int j = 0; j < vertices; j++)
            printf("%d ", graph[i][j]);

        printf("\n");
    }

    return 0;
}

2. Create the Graph as an Adjacency List with Operations Such as Insertion and Deletion

#include <stdio.h>
#include <stdlib.h>

#define MAX 50

struct Node {
    int vertex;
    struct Node *next;
};

struct Node *adj[MAX];
int vertices;

void insertEdge(int u, int v) {
    struct Node *n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->vertex = v;
    n->next = adj[u];
    adj[u] = n;

    n = malloc(sizeof(struct Node));

    if (n == NULL) {
        printf("Memory allocation failed\n");
        return;
    }

    n->vertex = u;
    n->next = adj[v];
    adj[v] = n;
}

void deleteOne(int u, int v) {
    struct Node *temp = adj[u], *prev = NULL;

    while (temp != NULL) {
        if (temp->vertex == v) {
            if (prev == NULL)
                adj[u] = temp->next;
            else
                prev->next = temp->next;

            free(temp);
            return;
        }

        prev = temp;
        temp = temp->next;
    }
}

void deleteEdge(int u, int v) {
    deleteOne(u, v);
    deleteOne(v, u);
}

void display() {
    printf("\nAdjacency List:\n");

    for (int i = 0; i < vertices; i++) {
        struct Node *temp = adj[i];

        printf("%d: ", i);

        while (temp != NULL) {
            printf("%d -> ", temp->vertex);
            temp = temp->next;
        }

        printf("NULL\n");
    }
}

int main() {
    int choice, u, v;

    printf("Enter number of vertices: ");
    scanf("%d", &vertices);

    if (vertices <= 0 || vertices > MAX) {
        printf("Invalid number of vertices\n");
        return 0;
    }

    for (int i = 0; i < vertices; i++)
        adj[i] = NULL;

    while (1) {
        printf("\n1. Insert Edge\n");
        printf("2. Delete Edge\n");
        printf("3. Display Graph\n");
        printf("4. Exit\n");
        printf("Enter choice: ");
        scanf("%d", &choice);

        if (choice == 4)
            break;

        if (choice == 1 || choice == 2) {
            printf("Enter two vertices: ");
            scanf("%d %d", &u, &v);

            if (u < 0 || u >= vertices ||
                v < 0 || v >= vertices) {
                printf("Invalid vertices\n");
                continue;
            }

            if (choice == 1)
                insertEdge(u, v);
            else
                deleteEdge(u, v);
        }
        else if (choice == 3) {
            display();
        }
        else {
            printf("Invalid choice\n");
        }
    }

    for (int i = 0; i < vertices; i++) {
        struct Node *temp = adj[i];

        while (temp != NULL) {
            struct Node *next = temp->next;
            free(temp);
            temp = next;
        }
    }

    return 0;
}

3. Traverse the Adjacency Matrix with Depth First Search

#include <stdio.h>

int graph[50][50];
int visited[50];
int vertices;

void DFS(int vertex) {
    visited[vertex] = 1;
    printf("%d ", vertex);

    for (int i = 0; i < vertices; i++) {
        if (graph[vertex][i] == 1 && !visited[i])
            DFS(i);
    }
}

int main() {
    int edges, u, v, start;

    printf("Enter number of vertices: ");
    scanf("%d", &vertices);

    if (vertices <= 0 || vertices > 50) {
        printf("Invalid number of vertices\n");
        return 0;
    }

    printf("Enter number of edges: ");
    scanf("%d", &edges);

    for (int i = 0; i < edges; i++) {
        printf("Enter edge (u v): ");
        scanf("%d %d", &u, &v);

        if (u < 0 || u >= vertices || v < 0 || v >= vertices) {
            printf("Invalid edge\n");
            i--;
            continue;
        }

        graph[u][v] = 1;
        graph[v][u] = 1;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    if (start < 0 || start >= vertices) {
        printf("Invalid starting vertex\n");
        return 0;
    }

    printf("DFS Traversal: ");
    DFS(start);

    for (int i = 0; i < vertices; i++) {
        if (!visited[i])
            DFS(i);
    }

    return 0;
}

5. Traverse the Adjacency Matrix with Breadth First Search

int visited[50];
int queue[50];
int vertices;

void BFS(int start) {
    int front = 0, rear = 0;

    visited[start] = 1;
    queue[rear++] = start;

    while (front < rear) {
        int vertex = queue[front++];
        printf("%d ", vertex);

        for (int i = 0; i < vertices; i++) {
            if (graph[vertex][i] == 1 && !visited[i]) {
                visited[i] = 1;
                queue[rear++] = i;

}
