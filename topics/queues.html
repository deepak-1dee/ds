1. Implement the operations of simple queue in an integer array (Insert, Delete, Peek, IsEmpty, IsFull)

#include <stdio.h>
#define MAX 100

int q[MAX], front = 0, rear = -1;

int isEmpty() {
    return rear == -1;
}

int isFull() {
    return rear == MAX - 1;
}

void insert(int x) {
    if (isFull()) {
        printf("Queue is full\n");
        return;
    }
    q[++rear] = x;
}

void delete() {
    if (isEmpty()) {
        printf("Queue is empty\n");
        return;
    }
    printf("Deleted: %d\n", q[front++]);

    if (front > rear) {
        front = 0;
        rear = -1;
    }
}

void peek() {
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Front: %d\n", q[front]);
}

int main() {
    insert(10);
    insert(20);
    insert(30);
    peek();
    delete();
    peek();
    return 0;
}

7. Implement the operations of priority queue in an integer array (Insert, Delete, Peek, IsEmpty, IsFull)

#include <stdio.h>
#define MAX 100

int data[MAX], priority[MAX], n = 0;

int isEmpty() {
    return n == 0;
}

int isFull() {
    return n == MAX;
}

void insert(int x, int p) {
    if (isFull()) {
        printf("Queue is full\n");
        return;
    }

    data[n] = x;
    priority[n] = p;
    n++;
}

int highest() {
    int i, pos = 0;

    for (i = 1; i < n; i++)
        if (priority[i] > priority[pos])
            pos = i;

    return pos;
}

void delete() {
    if (isEmpty()) {
        printf("Queue is empty\n");
        return;
    }

    int pos = highest();
    printf("Deleted: %d\n", data[pos]);

    for (int i = pos; i < n - 1; i++) {
        data[i] = data[i + 1];
        priority[i] = priority[i + 1];
    }

    n--;
}

void peek() {
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Highest priority element: %d\n", data[highest()]);
}

int main() {
    insert(10, 2);
    insert(20, 5);
    insert(30, 1);
    peek();
    delete();
    peek();
    return 0;
}

11. Implement the operations of priority queue dynamically of integers (Insert, Delete, Peek, IsEmpty, IsFull)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data, priority;
    struct Node *next;
};

struct Node *front = NULL;

int isEmpty() {
    return front == NULL;
}

int isFull() {
    struct Node *temp = malloc(sizeof(struct Node));
    if (temp == NULL)
        return 1;
    free(temp);
    return 0;
}

void insert(int x, int p) {
    struct Node *newNode = malloc(sizeof(struct Node));

    if (newNode == NULL) {
        printf("Memory full\n");
        return;
    }

    newNode->data = x;
    newNode->priority = p;

    if (front == NULL || p > front->priority) {
        newNode->next = front;
        front = newNode;
    } else {
        struct Node *temp = front;

        while (temp->next != NULL &&
               temp->next->priority >= p)
            temp = temp->next;

        newNode->next = temp->next;
        temp->next = newNode;
    }
}

void delete() {
    if (isEmpty()) {
        printf("Queue is empty\n");
        return;
    }

    struct Node *temp = front;
    printf("Deleted: %d\n", temp->data);
    front = front->next;
    free(temp);
}

void peek() {
    if (isEmpty())
        printf("Queue is empty\n");
    else
        printf("Highest priority element: %d\n", front->data);
}

int main() {
    insert(10, 2);
    insert(20, 5);
    insert(30, 1);
    peek();
    delete();
    peek();
    return 0;
}

13. Find the first circular tour that visits all petrol pumps

#include <stdio.h>

struct Pump {
    int petrol;
    int distance;
};

int circularTour(struct Pump p[], int n) {
    int start = 0, balance = 0, deficit = 0;

    for (int i = 0; i < n; i++) {
        balance += p[i].petrol - p[i].distance;

        if (balance < 0) {
            start = i + 1;
            deficit += balance;
            balance = 0;
        }
    }

    if (balance + deficit >= 0 && start < n)
        return start;

    return -1;
}

int main() {
    struct Pump p[] = {
        {6, 4},
        {3, 6},
        {7, 3}
    };

    int n = sizeof(p) / sizeof(p[0]);
    int result = circularTour(p, n);

    if (result == -1)
        printf("No possible tour\n");
    else
        printf("Starting petrol pump index: %d\n", result);

    return 0;
}

14. Find the length of the longest valid substring

#include <stdio.h>
#include <string.h>

int longestValid(char s[]) {
    int stack[1000];
    int top = -1;
    int lastInvalid = -1;
    int maxLength = 0;

    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] == '(') {
            stack[++top] = i;
        } else {
            if (top == -1) {
                lastInvalid = i;
            } else {
                top--;

                int length;

                if (top == -1)
                    length = i - lastInvalid;
                else
                    length = i - stack[top];

                if (length > maxLength)
                    maxLength = length;
            }
        }
    }

    return maxLength;
}

int main() {
    char s[1000];

    printf("Enter parentheses string: ");
    scanf("%999s", s);

    printf("Longest valid substring length: %d\n",
           longestValid(s));

    return 0;
}

15. Find the Next Greater Element

#include <stdio.h>

int main() {
    int a[100], result[100], stack[100];
    int n, top = -1;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter array elements: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    for (int i = n - 1; i >= 0; i--) {
        while (top != -1 && stack[top] <= a[i])
            top--;

        if (top == -1)
            result[i] = -1;
        else
            result[i] = stack[top];

        stack[++top] = a[i];
    }

    printf("Next Greater Elements:\n");
    for (int i = 0; i < n; i++)
        printf("%d -> %d\n", a[i], result[i]);

    return 0;
}

17. Queue based approach for first non-repeating character in a stream

#include <stdio.h>

int main() {
    char stream[1000], queue[1000];
    int count[256] = {0};
    int front = 0, rear = -1;

    printf("Enter character stream: ");
    scanf("%999s", stream);

    printf("First non-repeating characters:\n");

    for (int i = 0; stream[i] != '\0'; i++) {
        unsigned char ch = stream[i];

        count[ch]++;
        queue[++rear] = ch;

        while (front <= rear &&
               count[(unsigned char)queue[front]] > 1)
            front++;

        if (front <= rear)
            printf("%c ", queue[front]);
        else
            printf("-1 ");

    }

    printf("\n");
    return 0;
}

18. Reverse First K elements of Queue

#include <stdio.h>

#define MAX 1000

int main() {
    int q[MAX], stack[MAX];
    int n, k, top = -1;

    printf("Enter queue size: ");
    scanf("%d", &n);

    printf("Enter queue elements: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &q[i]);

    printf("Enter K: ");
    scanf("%d", &k);

    if (k < 0 || k > n) {
        printf("Invalid K\n");
        return 0;
    }

    for (int i = 0; i < k; i++)
        stack[++top] = q[i];

    for (int i = 0; i < k; i++)
        q[i] = stack[top--];

    printf("Reversed queue: ");

    for (int i = k; i < n; i++)
        printf("%d ", q[i]);

    printf("\n");

    return 0;
}

20. Rotten Oranges

#include <stdio.h>

#define MAX 100

int main() {
    int grid[MAX][MAX];
    int qr[MAX * MAX], qc[MAX * MAX];
    int rows, cols;
    int front = 0, rear = 0;
    int fresh = 0, minutes = 0;

    printf("Enter rows and columns: ");
    scanf("%d %d", &rows, &cols);

    if (rows < 1 || rows > MAX || cols < 1 || cols > MAX) {
        printf("Invalid grid size\n");
        return 0;
    }

    printf("Enter grid values (0, 1, 2):\n");

    for (int i = 0; i < rows; i++) {
        for (int j = 0; j < cols; j++) {
            scanf("%d", &grid[i][j]);

            if (grid[i][j] == 1)
                fresh++;

            if (grid[i][j] == 2) {
                qr[rear] = i;
                qc[rear] = j;
                rear++;
            }
        }
    }

    int dr[] = {-1, 1, 0, 0};
    int dc[] = {0, 0, -1, 1};

    while (front < rear && fresh > 0) {
        int size = rear - front;

        for (int i = 0; i < size; i++) {
            int r = qr[front];
            int c = qc[front];
            front++;

            for (int d = 0; d < 4; d++) {
                int nr = r + dr[d];
                int nc = c + dc[d];

                if (nr >= 0 && nr < rows &&
                    nc >= 0 && nc < cols &&
                    grid[nr][nc] == 1) {

                    grid[nr][nc] = 2;
                    fresh--;

                    qr[rear] = nr;
                    qc[rear] = nc;
                    rear++;
                }
            }
        }

        minutes++;
    }

    if (fresh == 0)
        printf("Minimum minutes: %d\n", minutes);
    else
        printf("All oranges cannot rot\n");

    return 0;
}
