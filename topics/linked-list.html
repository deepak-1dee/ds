1. Implement the operations of Single Linked List storing Integer Values (Insert at Begin, Insert at Pos, Insert at End, Delete at Begin, Delete at End, Delete at Pos, IsEmpty, IsFull, Display, Search)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *head = NULL;

int isEmpty() {
    return head == NULL;
}

int isFull() {
    struct Node *temp = malloc(sizeof(struct Node));
    if (temp == NULL)
        return 1;
    free(temp);
    return 0;
}

void insertBegin(int x) {
    struct Node *newNode = malloc(sizeof(struct Node));
    if (newNode == NULL) return;

    newNode->data = x;
    newNode->next = head;
    head = newNode;
}

void insertEnd(int x) {
    struct Node *newNode = malloc(sizeof(struct Node));
    if (newNode == NULL) return;

    newNode->data = x;
    newNode->next = NULL;

    if (head == NULL) {
        head = newNode;
        return;
    }

    struct Node *temp = head;
    while (temp->next != NULL)
        temp = temp->next;

    temp->next = newNode;
}

void insertPos(int x, int pos) {
    if (pos < 1) return;

    if (pos == 1) {
        insertBegin(x);
        return;
    }

    struct Node *temp = head;

    for (int i = 1; temp != NULL && i < pos - 1; i++)
        temp = temp->next;

    if (temp == NULL) {
        printf("Invalid position\n");
        return;
    }

    struct Node *newNode = malloc(sizeof(struct Node));
    if (newNode == NULL) return;

    newNode->data = x;
    newNode->next = temp->next;
    temp->next = newNode;
}

void deleteBegin() {
    if (isEmpty()) {
        printf("List is empty\n");
        return;
    }

    struct Node *temp = head;
    head = head->next;
    free(temp);
}

void deleteEnd() {
    if (isEmpty()) {
        printf("List is empty\n");
        return;
    }

    if (head->next == NULL) {
        free(head);
        head = NULL;
        return;
    }

    struct Node *temp = head;

    while (temp->next->next != NULL)
        temp = temp->next;

    free(temp->next);
    temp->next = NULL;
}

void deletePos(int pos) {
    if (pos < 1 || isEmpty()) {
        printf("Invalid position or empty list\n");
        return;
    }

    if (pos == 1) {
        deleteBegin();
        return;
    }

    struct Node *temp = head;

    for (int i = 1; temp->next != NULL && i < pos - 1; i++)
        temp = temp->next;

    if (temp->next == NULL) {
        printf("Invalid position\n");
        return;
    }

    struct Node *del = temp->next;
    temp->next = del->next;
    free(del);
}

void display() {
    struct Node *temp = head;

    if (isEmpty()) {
        printf("List is empty\n");
        return;
    }

    while (temp != NULL) {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

void search(int x) {
    struct Node *temp = head;
    int pos = 1;

    while (temp != NULL) {
        if (temp->data == x) {
            printf("Found at position %d\n", pos);
            return;
        }
        temp = temp->next;
        pos++;
    }

    printf("Not found\n");
}

int main() {
    insertBegin(20);
    insertBegin(10);
    insertEnd(40);
    insertPos(30, 3);

    display();
    search(30);

    deleteBegin();
    deleteEnd();
    deletePos(1);

    display();

    printf("Is Empty: %d\n", isEmpty());
    printf("Is Full: %d\n", isFull());

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int coeff, exp;
    struct Node *next;
};

struct Node *createTerm(int coeff, int exp) {
    struct Node *newNode = malloc(sizeof(struct Node));

    if (newNode == NULL)
        return NULL;

    newNode->coeff = coeff;
    newNode->exp = exp;
    newNode->next = NULL;

    return newNode;
}

void insertTerm(struct Node **head, int coeff, int exp) {
    if (coeff == 0) return;

    struct Node *temp = *head;

    while (temp != NULL) {
        if (temp->exp == exp) {
            temp->coeff += coeff;
            return;
        }
        temp = temp->next;
    }

    struct Node *newNode = createTerm(coeff, exp);
    if (newNode == NULL) return;

    if (*head == NULL || (*head)->exp < exp) {
        newNode->next = *head;
        *head = newNode;
        return;
    }

    temp = *head;

    while (temp->next != NULL && temp->next->exp > exp)
        temp = temp->next;

    newNode->next = temp->next;
    temp->next = newNode;
}

void readPolynomial(struct Node **head) {
    int n, coeff, exp;

    printf("Enter number of terms: ");
    scanf("%d", &n);

    for (int i = 0; i < n; i++) {
        printf("Enter coefficient and exponent: ");
        scanf("%d %d", &coeff, &exp);
        insertTerm(head, coeff, exp);
    }
}

struct Node *addPolynomial(struct Node *p1, struct Node *p2) {
    struct Node *result = NULL;

    while (p1 != NULL) {
        insertTerm(&result, p1->coeff, p1->exp);
        p1 = p1->next;
    }

    while (p2 != NULL) {
        insertTerm(&result, p2->coeff, p2->exp);
        p2 = p2->next;
    }

    return result;
}

        printf("%dx^%d", p->coeff, p->exp);

            printf(" + ");

        p = p->next;
    }

    printf("\n");
}
