1. Enter a matrix of 4 rows and 2 columns and display it in matrix format.

#include <stdio.h>
int main() {
    int a[4][2], i, j;
    printf("Enter 8 elements for a 4 x 2 matrix:\n");
    for (i = 0; i < 4; i++)
        for (j = 0; j < 2; j++)
            scanf("%d", &a[i][j]);

    printf("Matrix:\n");
    for (i = 0; i < 4; i++) {
        for (j = 0; j < 2; j++)
            printf("%d\t", a[i][j]);
        printf("\n");
    }
    return 0;
}

2. Enter two matrices from the user, add them, and display the result.

#include <stdio.h>
int main() {
    int a[10][10], b[10][10], sum[10][10];
    int rows, cols, i, j;
    printf("Enter rows and columns (max 10): ");
    scanf("%d%d", &rows, &cols);
    if (rows < 1 || rows > 10 || cols < 1 || cols > 10) {
        printf("Invalid matrix size.\n");
        return 0;
    }
    printf("Enter first matrix:\n");
    for (i = 0; i < rows; i++)
        for (j = 0; j < cols; j++) scanf("%d", &a[i][j]);
    printf("Enter second matrix:\n");
    for (i = 0; i < rows; i++)
        for (j = 0; j < cols; j++) scanf("%d", &b[i][j]);
    for (i = 0; i < rows; i++)
        for (j = 0; j < cols; j++) sum[i][j] = a[i][j] + b[i][j];
    printf("Sum:\n");
    for (i = 0; i < rows; i++) {
        for (j = 0; j < cols; j++) printf("%d\t", sum[i][j]);
        printf("\n");
    }
    return 0;
}

4. Enter two matrices, multiply them, and display the result.

#include <stdio.h>
int main() {
    int a[10][10], b[10][10], product[10][10] = {0};
    int r1, c1, r2, c2, i, j, k;
    printf("Enter rows and columns of first matrix: ");
    scanf("%d%d", &r1, &c1);
    printf("Enter rows and columns of second matrix: ");
    scanf("%d%d", &r2, &c2);
    if (r1 < 1 || r1 > 10 || c1 < 1 || c1 > 10 ||
        r2 < 1 || r2 > 10 || c2 < 1 || c2 > 10 || c1 != r2) {
        printf("Invalid sizes. Columns of first matrix must equal rows of second.\n");
        return 0;
    }
    printf("Enter first matrix:\n");
    for (i = 0; i < r1; i++)
        for (j = 0; j < c1; j++) scanf("%d", &a[i][j]);
    printf("Enter second matrix:\n");
    for (i = 0; i < r2; i++)
        for (j = 0; j < c2; j++) scanf("%d", &b[i][j]);
    for (i = 0; i < r1; i++)
        for (j = 0; j < c2; j++)
            for (k = 0; k < c1; k++) product[i][j] += a[i][k] * b[k][j];
    printf("Product:\n");
    for (i = 0; i < r1; i++) {
        for (j = 0; j < c2; j++) printf("%d\t", product[i][j]);
        printf("\n");
    }
    return 0;
}

6. Enter a matrix, multiply it by a scalar value, and display the result.

#include <stdio.h>
int main() {
    int a[10][10], result[10][10], rows, cols, scalar, i, j;
    printf("Enter rows and columns (max 10): ");

    if (rows < 1 || rows > 10 || cols < 1 || cols > 10) {
        printf("Invalid matrix size.\n");
        return 0;
    }
    printf("Enter matrix elements:\n");
    for (i = 0; i < rows; i++)
        for (j = 0; j < cols; j++) scanf("%d", &a[i][j]);
    printf("Enter scalar: ");
    scanf("%d", &scalar);
    for (i = 0; i < rows; i++)
        for (j = 0; j < cols; j++) result[i][j] = a[i][j] * scalar;
    printf("Result:\n");
    for (i = 0; i < rows; i++) {
        for (j = 0; j < cols; j++) printf("%d\t", result[i][j]);
        printf("\n");
    }
    return 0;
}

7. Create a two-player Tic-Tac-Toe game using a 3 x 3 two-dimensional array.

#include <stdio.h>
int main() {
    char board[3][3] = {{'1','2','3'}, {'4','5','6'}, {'7','8','9'}};
    int player = 1, choice, row, col, moves = 0, i;
    char mark;
    while (moves < 9) {
        printf("\n");
        for (i = 0; i < 3; i++)
            printf(" %c | %c | %c\n---+---+---\n", board[i][0], board[i][1], board[i][2]);
        printf("Player %d, choose a cell (1-9): ", player);
        if (scanf("%d", &choice) != 1) return 0;
        if (choice < 1 || choice > 9) {
            printf("Invalid choice. Try again.\n");
            continue;
        }
        row = (choice - 1) / 3;
        col = (choice - 1) % 3;
        if (board[row][col] == 'X' || board[row][col] == 'O') {
            printf("That cell is already taken. Try again.\n");
            continue;
        }
        mark = (player == 1) ? 'X' : 'O';
        board[row][col] = mark;
        moves++;
        if ((board[row][0] == mark && board[row][1] == mark && board[row][2] == mark) ||
            (board[0][col] == mark && board[1][col] == mark && board[2][col] == mark) ||
            (row == col && board[0][0] == mark && board[1][1] == mark && board[2][2] == mark) ||
            (row + col == 2 && board[0][2] == mark && board[1][1] == mark && board[2][0] == mark)) {
            printf("\n");
            for (i = 0; i < 3; i++)
                printf(" %c | %c | %c\n", board[i][0], board[i][1], board[i][2]);
            printf("Player %d wins!\n", player);
            return 0;
        }
        player = (player == 1) ? 2 : 1;
    }
    printf("\n");
    for (i = 0; i < 3; i++)
        printf(" %c | %c | %c\n", board[i][0], board[i][1], board[i][2]);
    printf("It's a draw!\n");
    return 0;
}
