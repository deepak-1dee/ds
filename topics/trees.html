#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

int pre[100], in[100], index = 0;

struct Node *create(int start, int end) {
    if (start > end) return NULL;

    struct Node *root = malloc(sizeof(struct Node));
    root->data = pre[index++];
    root->left = root->right = NULL;

    int i;
    for (i = start; i <= end; i++)
        if (in[i] == root->data) break;

    root->left = create(start, i - 1);
    root->right = create(i + 1, end);

    return root;
}

void inorder(struct Node *root) {
    if (root == NULL) return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

int main() {
    int n;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter preorder traversal: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &pre[i]);

    printf("Enter inorder traversal: ");
    for (int i = 0; i < n; i++)
        scanf("%d", &in[i]);

    struct Node *root = create(0, n - 1);

    printf("Constructed tree inorder: ");
    inorder(root);

    return 0;
}

2. Binary Tree Level Order Traversal

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *create(int x) {
    struct Node *newNode = malloc(sizeof(struct Node));
    newNode->data = x;
    newNode->left = newNode->right = NULL;
    return newNode;
}

void levelOrder(struct Node *root) {
    if (root == NULL) return;

    struct Node *queue[1000];
    int front = 0, rear = 0;

    queue[rear++] = root;

    while (front < rear) {
        struct Node *temp = queue[front++];
        printf("%d ", temp->data);

        if (temp->left != NULL)
            queue[rear++] = temp->left;

        if (temp->right != NULL)
            queue[rear++] = temp->right;
    }
}

int main() {
    struct Node *root = create(1);
    root->left = create(2);
    root->right = create(3);
    root->left->left = create(4);
    root->left->right = create(5);
    root->right->left = create(6);

    printf("Level Order Traversal: ");
    levelOrder(root);

    return 0;
}

6. Find the Maximum Depth of Binary Tree

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *create(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->data = x;
    n->left = n->right = NULL;
    return n;
}

int maxDepth(struct Node *root) {
    if (root == NULL)
        return 0;

    int left = maxDepth(root->left);
    int right = maxDepth(root->right);

    return 1 + (left > right ? left : right);
}

    root->right = create(3);
    root->left->left = create(4);
    root->left->right = create(5);

    printf("Maximum Depth: %d\n", maxDepth(root));

    return 0;
}

7. Create Binary Search Tree Using Arrays (Insert, Delete, Search, Preorder, Inorder, Postorder, Minimum Element)

#include <stdio.h>
#define MAX 1000

int tree[MAX];

void insert(int x) {
    int i = 0;

    while (i < MAX) {
        if (tree[i] == -1) {
            tree[i] = x;
            return;
        }

        if (x < tree[i])
            i = 2 * i + 1;
        else if (x > tree[i])
            i = 2 * i + 2;
        else
            return;
    }

    printf("Array is full\n");
}

int search(int x) {
    int i = 0;

    while (i < MAX && tree[i] != -1) {
        if (tree[i] == x) return i;

        if (x < tree[i])
            i = 2 * i + 1;
        else
            i = 2 * i + 2;
    }

    return -1;
}

void inorder(int i) {
    if (i >= MAX || tree[i] == -1) return;
    inorder(2 * i + 1);
    printf("%d ", tree[i]);
    inorder(2 * i + 2);
}

void preorder(int i) {
    if (i >= MAX || tree[i] == -1) return;
    printf("%d ", tree[i]);
    preorder(2 * i + 1);
    preorder(2 * i + 2);
}

void postorder(int i) {
    if (i >= MAX || tree[i] == -1) return;
    postorder(2 * i + 1);
    postorder(2 * i + 2);
    printf("%d ", tree[i]);
}

int minimum() {
    int i = 0;

    if (tree[i] == -1) return -1;

    while (2 * i + 1 < MAX && tree[2 * i + 1] != -1)
        i = 2 * i + 1;

    return tree[i];
}

int maximumValue(int i) {
    if (i >= MAX || tree[i] == -1) return -1;

    int a = maximumValue(2 * i + 1);
    int b = maximumValue(2 * i + 2);

    int max = tree[i];
    if (a > max) max = a;
    if (b > max) max = b;

    return max;
}

void deleteNode(int x) {
    int pos = search(x);

    if (pos == -1) {
        printf("Not found\n");
        return;
    }

    int max = maximumValue(pos);

    if (max == x) {
        tree[pos] = -1;
        return;
    }

    int min = tree[pos];
    int replacement = -1;

    for (int i = 0; i < MAX; i++) {
        if (tree[i] == -1) continue;

        if (tree[i] > min && (replacement == -1 || tree[i] < replacement))
            replacement = tree[i];
    }

    if (replacement == -1) {
        printf("Cannot delete this node in array representation\n");
        return;
    }

    tree[pos] = replacement;
    deleteNode(replacement);
}

int main() {
    for (int i = 0; i < MAX; i++)
        tree[i] = -1;

    insert(50);
    insert(30);
    insert(70);
    insert(20);
    insert(40);
    insert(60);
    insert(80);

    printf("Inorder: ");
    inorder(0);

    printf("\nPreorder: ");
    preorder(0);

    printf("\nPostorder: ");
    postorder(0);

    printf("\nMinimum: %d\n", minimum());

    printf("Search 40: %s\n", search(40) != -1 ? "Found" : "Not Found");

    deleteNode(30);

    printf("After deletion: ");
    inorder(0);

    return 0;
}

8. Create Binary Search Tree Dynamically (Insert, Delete, Search, Preorder, Inorder, Postorder, Minimum Element)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        root = malloc(sizeof(struct Node));
        root->data = x;
        root->left = root->right = NULL;
        return root;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

struct Node *minimumNode(struct Node *root) {
    while (root != NULL && root->left != NULL)
        root = root->left;
    return root;
}

struct Node *deleteNode(struct Node *root, int x) {
    if (root == NULL) return NULL;

    if (x < root->data)
        root->left = deleteNode(root->left, x);
    else if (x > root->data)
        root->right = deleteNode(root->right, x);
    else {
        if (root->left == NULL) {
            struct Node *temp = root->right;
            free(root);
            return temp;
        }

        if (root->right == NULL) {
            struct Node *temp = root->left;
            free(root);
            return temp;
        }

        struct Node *temp = minimumNode(root->right);
        root->data = temp->data;
        root->right = deleteNode(root->right, temp->data);
    }

    return root;
}

int search(struct Node *root, int x) {
    if (root == NULL) return 0;
    if (root->data == x) return 1;

    if (x < root->data)
        return search(root->left, x);

    return search(root->right, x);
}

void inorder(struct Node *root) {
    if (root == NULL) return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

void preorder(struct Node *root) {
    if (root == NULL) return;
    printf("%d ", root->data);
    preorder(root->left);
    preorder(root->right);
}

void postorder(struct Node *root) {
    if (root == NULL) return;
    postorder(root->left);
    postorder(root->right);
    printf("%d ", root->data);
}

int main() {
    struct Node *root = NULL;

    root = insert(root, 50);
    root = insert(root, 30);
    root = insert(root, 70);
    root = insert(root, 20);
    root = insert(root, 40);
    root = insert(root, 60);
    root = insert(root, 80);

    printf("Inorder: ");
    inorder(root);

    printf("\nPreorder: ");
    preorder(root);

    printf("\nPostorder: ");
    postorder(root);

    struct Node *m = minimumNode(root);
    if (m != NULL)
        printf("\nMinimum: %d", m->data);

    printf("\nSearch 40: %s", search(root, 40) ? "Found" : "Not Found");

    printf("\nAfter deletion: ");
    inorder(root);

9. Find Inorder Predecessor and Inorder Successor of Binary Search Tree (Dynamically Represented)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *n = malloc(sizeof(struct Node));
        n->data = x;
        n->left = n->right = NULL;
        return n;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

void findPreSuc(struct Node *root, int key,
                struct Node **pre, struct Node **suc) {
    *pre = NULL;
    *suc = NULL;

    struct Node *current = root;

    while (current != NULL) {
        if (current->data == key) {
            if (current->left != NULL) {
                struct Node *t = current->left;
                while (t->right != NULL)
                    t = t->right;
                *pre = t;
            }

            if (current->right != NULL) {
                struct Node *t = current->right;
                while (t->left != NULL)
                    t = t->left;
                *suc = t;
            }

            return;
        }

        if (key < current->data) {
            *suc = current;
            current = current->left;
        } else {
            *pre = current;
            current = current->right;
        }
    }
}

int main() {
    struct Node *root = NULL;

    root = insert(root, 50);
    root = insert(root, 30);
    root = insert(root, 70);
    root = insert(root, 20);
    root = insert(root, 40);
    root = insert(root, 60);
    root = insert(root, 80);

    int key = 50;
    struct Node *pre, *suc;

    findPreSuc(root, key, &pre, &suc);

    if (pre != NULL)
        printf("Predecessor: %d\n", pre->data);
    else
        printf("No predecessor\n");

    if (suc != NULL)
        printf("Successor: %d\n", suc->data);
    else
        printf("No successor\n");

    return 0;
}

10. Check Whether Binary Search Tree Contains Dead End

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *n = malloc(sizeof(struct Node));
        n->data = x;
        n->left = n->right = NULL;
        return n;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

int deadEnd(struct Node *root, int min, int max) {
    if (root == NULL) return 0;

    if (min == max)
        return 1;

    return deadEnd(root->left, min, root->data - 1) ||
           deadEnd(root->right, root->data + 1, max);
}

int main() {
    struct Node *root = NULL;

    root = insert(root, 8);
    root = insert(root, 5);
    root = insert(root, 2);
    root = insert(root, 7);
    root = insert(root, 9);

    if (deadEnd(root, 1, 100000))
        printf("BST contains a dead end\n");
    else
        printf("BST does not contain a dead end\n");

    return 0;
}

11. Create Binary Tree to Binary Search Tree Through Array

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

int a[100], sorted[100], n, count = 0;

struct Node *createTree(int i) {
    if (i >= n || a[i] == -1)
        return NULL;

    struct Node *root = malloc(sizeof(struct Node));
    root->data = a[i];

    root->left = createTree(2 * i + 1);
    root->right = createTree(2 * i + 2);

    return root;
}

void collect(struct Node *root) {
    if (root == NULL) return;
    collect(root->left);
    sorted[count++] = root->data;
    collect(root->right);
}

void sortArray() {
    for (int i = 0; i < count - 1; i++) {
        for (int j = 0; j < count - i - 1; j++) {
            if (sorted[j] > sorted[j + 1]) {
                int t = sorted[j];
                sorted[j] = sorted[j + 1];
                sorted[j + 1] = t;
            }
        }
    }
}

void convert(struct Node *root, int *index) {
    if (root == NULL) return;

    convert(root->left, index);
    root->data = sorted[(*index)++];
    convert(root->right, index);
}

void inorder(struct Node *root) {
    if (root == NULL) return;
    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

int main() {
    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter level order values (-1 for missing nodes):\n");
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    struct Node *root = createTree(0);

    collect(root);
    sortArray();

    int index = 0;
    convert(root, &index);

    printf("Binary Search Tree inorder: ");
    inorder(root);

    return 0;
}

    if (n <= 0 || n > 100) {
        printf("Invalid number of nodes\n");
        return 0;
    }

    printf("Enter level order node values:\n");

    scanf("%d", &x);
    struct Node *root = createNode(x);
    queue[rear++] = root;

    int created = 1;

        struct Node *temp = queue[front++];

        printf("Enter left child of %d: ", temp->data);
        scanf("%d", &x);

        temp->left = createNode(x);
        queue[rear++] = temp->left;
        created++;

        if (created == n) break;

        printf("Enter right child of %d: ", temp->data);
        scanf("%d", &x);

        temp->right = createNode(x);
        queue[rear++] = temp->right;
        created++;
    }

    printf("Almost complete binary tree created.\n");

    printf("Level order: ");
    for (int i = 0; i < rear; i++)
        printf("%d ", queue[i]->data);

19. Create Complete Binary Tree from the Given Input by the User (Dynamically)

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *createNode(int x) {
    struct Node *n = malloc(sizeof(struct Node));
    n->data = x;
    n->left = n->right = NULL;
    return n;
}

int main() {
    struct Node *queue[100];
    int n, x, front = 0, rear = 0;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    if (n <= 0 || n > 100) {
        printf("Invalid number of nodes\n");
        return 0;
    }

    printf("Enter root value: ");
    scanf("%d", &x);

    struct Node *root = createNode(x);
    queue[rear++] = root;

    for (int i = 1; i < n; i++) {
        struct Node *parent = queue[front];

        printf("Enter value for node %d: ", i + 1);
        scanf("%d", &x);

        struct Node *child = createNode(x);

        if (parent->left == NULL)
            parent->left = child;
        else {
            parent->right = child;
            front++;
        }

        queue[rear++] = child;
    }

    printf("Complete binary tree level order: ");

    for (int i = 0; i < rear; i++)
        printf("%d ", queue[i]->data);

    return 0;
}

20. Count the Number of Nodes in Binary Search Tree Constructed Dynamically

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left, *right;
};

struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *n = malloc(sizeof(struct Node));
        n->data = x;
        n->left = n->right = NULL;
        return n;
    }

    if (x < root->data)
        root->left = insert(root->left, x);
    else if (x > root->data)
        root->right = insert(root->right, x);

    return root;
}

int countNodes(struct Node *root) {
    if (root == NULL)
        return 0;

    return 1 + countNodes(root->left)
             + countNodes(root->right);
}

void inorder(struct Node *root) {
    if (root == NULL) return;

    inorder(root->left);
    printf("%d ", root->data);
    inorder(root->right);
}

int main() {
    struct Node *root = NULL;
    int n, x;

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("Enter BST values:\n");

    for (int i = 0; i < n; i++) {
        scanf("%d", &x);
        root = insert(root, x);
    }

    printf("BST inorder: ");
    inorder(root);

    printf("\nNumber of nodes: %d\n", countNodes(root));

    return 0;
}
