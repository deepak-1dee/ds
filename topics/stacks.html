1. Create an integer stack using an array with PUSH, POP, PEEK, ISEMPTY, and ISFULL operations.

#include <stdio.h>
#define MAX 100
int stack[MAX], top = -1;
int isEmpty() { return top == -1; }
int isFull() { return top == MAX - 1; }
void push(int value) {
    if (isFull()) printf("Stack is full.\n");
    else { stack[++top] = value; printf("%d pushed.\n", value); }
}
void pop() {
    if (isEmpty()) printf("Stack is empty.\n");
    else printf("Popped: %d\n", stack[top--]);
}
void peek() {
    if (isEmpty()) printf("Stack is empty.\n");
    else printf("Top: %d\n", stack[top]);
}
int main() {
    int choice, value;
    while (1) {
        printf("\n1.Push 2.Pop 3.Peek 4.IsEmpty 5.IsFull 6.Display 7.Exit\nChoice: ");
        scanf("%d", &choice);
        if (choice == 1) { printf("Value: "); scanf("%d", &value); push(value); }
        else if (choice == 2) pop();
        else if (choice == 3) peek();
        else if (choice == 4) printf(isEmpty() ? "Stack is empty.\n" : "Stack is not empty.\n");
        else if (choice == 5) printf(isFull() ? "Stack is full.\n" : "Stack is not full.\n");
        else if (choice == 6) {
            int i;
            if (isEmpty()) printf("Stack is empty.\n");
            else { for (i = top; i >= 0; i--) printf("%d\n", stack[i]); }
        } else if (choice == 7) break;
        else printf("Invalid choice.\n");
    }
    return 0;
}

3. Create a dynamically allocated integer-node stack with PUSH, POP, PEEK, ISEMPTY, and I SFULL operations.

#include <stdio.h>
#include <stdlib.h>
typedef struct Node { int data; struct Node *next; } Node;
Node *top = NULL;
int isEmpty() { return top == NULL; }
int isFull() {
    Node *p = (Node *)malloc(sizeof(Node));
    if (p == NULL) return 1;
    free(p); return 0;
}
void push(int value) {
    Node *p = (Node *)malloc(sizeof(Node));
    if (p == NULL) { printf("Stack is full (memory unavailable).\n"); return; }
    p->data = value; p->next = top; top = p;
    printf("%d pushed.\n", value);
}
void pop() {
    Node *p;
    if (isEmpty()) { printf("Stack is empty.\n"); return; }
    p = top; printf("Popped: %d\n", p->data); top = top->next; free(p);
}
void peek() {
    if (isEmpty()) printf("Stack is empty.\n");
    else printf("Top: %d\n", top->data);
}
int main() {
    int choice, value; Node *p;
    while (1) {
        printf("\n1.Push 2.Pop 3.Peek 4.IsEmpty 5.IsFull 6.Display 7.Exit\nChoice: ");
        scanf("%d", &choice);
        if (choice == 1) { printf("Value: "); scanf("%d", &value); push(value); }
        else if (choice == 2) pop();
        else if (choice == 3) peek();
        else if (choice == 4) printf(isEmpty() ? "Stack is empty.\n" : "Stack is not empty.\n");
        else if (choice == 5) printf(isFull() ? "Stack is full (memory unavailable).\n" : "Stack is not full.\n");
        else if (choice == 6) {
            for (p = top; p != NULL; p = p->next) printf("%d\n", p->data);
            if (isEmpty()) printf("Stack is empty.\n");
        } else if (choice == 7) break;
        else printf("Invalid choice.\n");
    }
    while (top != NULL) pop();
    return 0;
}

5. Enter a string and reverse it using a stack.

#include <stdio.h>
#include <string.h>
#define MAX 500
int main() {
    char s[MAX], stack[MAX]; int top = -1, i;
    printf("Enter a string: ");
    fgets(s, sizeof(s), stdin);
    s[strcspn(s, "\n")] = '\0';
    for (i = 0; s[i] != '\0'; i++) stack[++top] = s[i];
    printf("Reversed string: ");
    while (top >= 0) putchar(stack[top--]);
    printf("\n");
    return 0;
}

6. Match round parentheses using a stack (check whether every opening bracket has a matching closing bracket).

#include <stdio.h>
#include <string.h>
#define MAX 500
int main() {
    char s[MAX], stack[MAX]; int top = -1, i, valid = 1;
    printf("Enter an expression: ");
    fgets(s, sizeof(s), stdin);
    for (i = 0; s[i] != '\0'; i++) {
        if (s[i] == '(') stack[++top] = '(';
        else if (s[i] == ')') {
            if (top == -1) { valid = 0; break; }
            top--;
        }
    }
    if (top != -1) valid = 0;
    if (valid) printf("Parentheses are balanced.\n");
    else printf("Parentheses are not balanced.\n");
    return 0;
}

7. Create two integer stacks in a single array.

#include <stdio.h>
#define MAX 100
int a[MAX], top1 = -1, top2 = MAX;
int main() {
    int choice, stackNo, value, i;
    while (1) {
        printf("\n1.Push 2.Pop 3.Display 4.Exit\nChoice: ");
        scanf("%d", &choice);
        if (choice == 4) break;
        if (choice == 1) {
            printf("Stack number (1 or 2): "); scanf("%d", &stackNo);
            printf("Value: "); scanf("%d", &value);
            if (top1 + 1 == top2) printf("Combined array is full.\n");
            else if (stackNo == 1) a[++top1] = value;
            else if (stackNo == 2) a[--top2] = value;
            else printf("Invalid stack number.\n");
        } else if (choice == 2) {
            printf("Stack number (1 or 2): "); scanf("%d", &stackNo);
            if (stackNo == 1) {
                if (top1 == -1) printf("Stack 1 is empty.\n");
                else printf("Popped from stack 1: %d\n", a[top1--]);
            } else if (stackNo == 2) {
                if (top2 == MAX) printf("Stack 2 is empty.\n");
                else printf("Popped from stack 2: %d\n", a[top2++]);
            } else printf("Invalid stack number.\n");
        } else if (choice == 3) {
            printf("Stack 1: ");
            for (i = top1; i >= 0; i--) printf("%d ", a[i]);
            printf("\nStack 2: ");
            for (i = top2; i < MAX; i++) printf("%d ", a[i]);
            printf("\n");
        } else printf("Invalid choice.\n");
    }
    return 0;
}

8. Sort an integer stack in ascending order using an array.

#include <stdio.h>
#define MAX 100
int main() {
    int s[MAX], temp[MAX], n, i, j, x, top = -1, ttop = -1;
    printf("Enter number of elements (1-100): "); scanf("%d", &n);
    if (n < 1 || n > MAX) { printf("Invalid size.\n"); return 0; }
    printf("Enter stack elements from bottom to top: ");
    for (i = 0; i < n; i++) { scanf("%d", &x); s[++top] = x; }
    while (top >= 0) {
        x = s[top--];
        while (ttop >= 0 && temp[ttop] > x) s[++top] = temp[ttop--];
        temp[++ttop] = x;
    }
    printf("Stack sorted in ascending order (bottom to top): ");
    for (i = 0; i <= ttop; i++) printf("%d ", temp[i]);
    printf("\n");
    return 0;
}

9. Convert an infix expression into prefix notation (single-character operands; supports +, -, *, /, ^ and parentheses).

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 300
int prec(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    if (c == '+' || c == '-') return 1;
    return 0;
}
int main() {
    char infix[MAX], rev[MAX], stack[MAX], post[MAX], prefix[MAX];
    int i, j = 0, top = -1, k = 0, len;
    printf("Enter infix expression: "); fgets(infix, sizeof(infix), stdin);
    infix[strcspn(infix, "\n")] = '\0'; len = (int)strlen(infix);
    for (i = len - 1; i >= 0; i--) {
        if (infix[i] == '(') rev[j++] = ')';
        else if (infix[i] == ')') rev[j++] = '(';
        else if (!isspace((unsigned char)infix[i])) rev[j++] = infix[i];
    }
    rev[j] = '\0';
    for (i = 0; rev[i] != '\0'; i++) {
        char c = rev[i];
        if (isalnum((unsigned char)c)) post[k++] = c;
        else if (c == '(') stack[++top] = c;
        else if (c == ')') {
            while (top >= 0 && stack[top] != '(') post[k++] = stack[top--];
            if (top >= 0) top--;
        } else {
            while (top >= 0 && stack[top] != '(' &&
                  (prec(stack[top]) > prec(c) ||
                  (prec(stack[top]) == prec(c) && c == '^')))
                post[k++] = stack[top--];
            stack[++top] = c;
        }
    }
    while (top >= 0) post[k++] = stack[top--];
    post[k] = '\0';
    for (i = k - 1; i >= 0; i--) prefix[k - 1 - i] = post[i];
    prefix[k] = '\0';
    printf("Prefix expression: %s\n", prefix);
    return 0;
}

10. Convert an infix expression into postfix notation (single-character operands; supports +, -, *, /, ^ and parentheses).

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 300
int prec(char c) {
    if (c == '^') return 3;
    if (c == '*' || c == '/') return 2;
    if (c == '+' || c == '-') return 1;
    return 0;
}
int main() {
    char s[MAX], stack[MAX], out[MAX]; int i, top = -1, k = 0;
    printf("Enter infix expression: "); fgets(s, sizeof(s), stdin);
    for (i = 0; s[i] != '\0' && s[i] != '\n'; i++) {
        char c = s[i];
        if (isspace((unsigned char)c)) continue;
        if (isalnum((unsigned char)c)) out[k++] = c;
        else if (c == '(') stack[++top] = c;
        else if (c == ')') {
            while (top >= 0 && stack[top] != '(') out[k++] = stack[top--];
            if (top >= 0) top--;
        } else {
            while (top >= 0 && stack[top] != '(' &&
                  (prec(stack[top]) > prec(c) ||
                  (prec(stack[top]) == prec(c) && c != '^')))
                out[k++] = stack[top--];
            stack[++top] = c;
        }
    }
    while (top >= 0) out[k++] = stack[top--];
    out[k] = '\0';
    printf("Postfix expression: %s\n", out);
    return 0;
}

11. Evaluate a prefix expression containing single-digit non-negative operands and operators +, -, *, and /.

#include <stdio.h>
#include <string.h>
#include <ctype.h>
#define MAX 300
int main() {
    char s[MAX]; int st[MAX], top = -1, i;
    printf("Enter prefix expression (single digits, no spaces): ");
    scanf("%299s", s);
    for (i = (int)strlen(s) - 1; i >= 0; i--) {
        char c = s[i];
        if (isdigit((unsigned char)c)) st[++top] = c - '0';
        else if (c == '+' || c == '-' || c == '*' || c == '/') {
            int a, b;
            if (top < 1) { printf("Invalid expression.\n"); return 0; }
            a = st[top--]; b = st[top--];
            if (c == '+') st[++top] = a + b;
            else if (c == '-') st[++top] = a - b;
            else if (c == '*') st[++top] = a * b;
            else {
                if (b == 0) { printf("Division by zero.\n"); return 0; }
                st[++top] = a / b;
            }
        } else { printf("Unsupported character.\n"); return 0; }
    }
    if (top == 0) printf("Result = %d\n", st[top]);
    else printf("Invalid expression.\n");
    return 0;
}

13. Implement Tower of Hanoi using stacks to represent the three rods.

#include <stdio.h>
#define MAX 20
int rods[3][MAX], top[3] = {-1, -1, -1};
char names[3] = {'A', 'B', 'C'};
void push(int rod, int disk) { rods[rod][++top[rod]] = disk; }
int pop(int rod) { return rods[rod][top[rod]--]; }
void moveDisk(int from, int to) {
    int disk = pop(from);
    push(to, disk);
    printf("Move disk %d from rod %c to rod %c\n", disk, names[from], names[to]);
}
void hanoi(int n, int from, int to, int aux) {
    if (n == 1) { moveDisk(from, to); return; }
    hanoi(n - 1, from, aux, to);
    moveDisk(from, to);
    hanoi(n - 1, aux, to, from);
}
int main() {
    int n, i;
    printf("Enter number of disks (1-20): "); scanf("%d", &n);
    if (n < 1 || n > 20) { printf("Invalid number of disks.\n"); return 0; }
    for (i = n; i >= 1; i--) push(0, i);
    printf("Steps to solve Tower of Hanoi:\n");
    hanoi(n, 0, 2, 1);
    return 0;
}
