#include <stdio.h>
#include <time.h>

int c = 0;

void merge(int a[], int l, int m, int r) {
    int temp[100], i = l, j = m + 1, k = 0;

    while (i <= m && j <= r) {
        c++;
        if (a[i] <= a[j]) temp[k++] = a[i++];
        else temp[k++] = a[j++];
    }

    while (i <= m) temp[k++] = a[i++];
    while (j <= r) temp[k++] = a[j++];

    for (i = l, k = 0; i <= r; i++, k++)
        a[i] = temp[k];
}

int main() {
    int a[100], n;
    clock_t start, end;

    printf("Enter array size: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) return 0;

    for (int width = 1; width < n; width *= 2) {
        for (int l = 0; l < n; l += 2 * width) {
            int m = l + width - 1;
            int r = l + 2 * width - 1;
            if (m >= n) continue;
            if (r >= n) r = n - 1;
            merge(a, l, m, r);

    end = clock();

    printf("Sorted array: ");

9. Enter an Integer Array and Perform Quick Sort Through Iteration. Calculate Best and Worst Complexity with Time

#include <stdio.h>
#include <time.h>

int c = 0;

int partition(int a[], int low, int high) {
    int pivot = a[high], i = low - 1;

    for (int j = low; j < high; j++) {
        c++;
        if (a[j] <= pivot) {
            i++;
            int t = a[i]; a[i] = a[j]; a[j] = t;
        }
    }

    int t = a[i + 1];
    a[i + 1] = a[high];
    a[high] = t;

    return i + 1;
}

int main() {
    int a[100], stack[200], top = -1, n;
    clock_t start, end;

    printf("Enter array size: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) return 0;

    printf("Enter elements:\n");
    for (int i = 0; i < n; i++) scanf("%d", &a[i]);

    start = clock();

    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0) {
        int high = stack[top--];
        int low = stack[top--];

        if (low < high) {
            int p = partition(a, low, high);

            if (p - 1 > low) {
                stack[++top] = low;
                stack[++top] = p - 1;
            }
            if (p + 1 < high) {
                stack[++top] = p + 1;
                stack[++top] = high;
            }
        }
    }

    end = clock();

    printf("Sorted array: ");
    for (int i = 0; i < n; i++) printf("%d ", a[i]);

    printf("\nComparisons: %d", c);
    printf("\nTime: %.9f seconds", (double)(end - start) / CLOCKS_PER_SEC);
    printf("\nBest complexity: O(n log n)");
    printf("\nWorst complexity: O(n^2)\n");

    return 0;
}

17. Enter a String Array and Perform Merge Sort Through Iteration. Calculate Best and Worst Complexity with Time

#include <stdio.h>
#include <string.h>
#include <time.h>

int c = 0;

void merge(char a[][100], int l, int m, int r) {
    char temp[100][100];
    int i = l, j = m + 1, k = 0;

    while (i <= m && j <= r) {
        c++;
        if (strcmp(a[i], a[j]) <= 0)
            strcpy(temp[k++], a[i++]);
        else
            strcpy(temp[k++], a[j++]);
    }

    while (i <= m) strcpy(temp[k++], a[i++]);
    while (j <= r) strcpy(temp[k++], a[j++]);

    for (i = l, k = 0; i <= r; i++, k++)
        strcpy(a[i], temp[k]);
}

int main() {
    char a[100][100];
    int n;
    clock_t start, end;

    printf("Enter number of strings: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) return 0;

    printf("Enter strings:\n");
    for (int i = 0; i < n; i++) scanf("%99s", a[i]);

    start = clock();

    for (int width = 1; width < n; width *= 2) {
        for (int l = 0; l < n; l += 2 * width) {
            int m = l + width - 1;
            int r = l + 2 * width - 1;
            if (m >= n) continue;
            if (r >= n) r = n - 1;
            merge(a, l, m, r);
        }
    }

    end = clock();

    printf("Sorted strings:\n");
    for (int i = 0; i < n; i++) printf("%s\n", a[i]);

    printf("Comparisons: %d", c);
    printf("\nTime: %.9f seconds", (double)(end - start) / CLOCKS_PER_SEC);
    printf("\nBest complexity: O(n log n)");
    printf("\nWorst complexity: O(n log n)\n");

#include <stdio.h>
#include <string.h>
#include <time.h>

int c = 0;

void merge(char a[][100], int l, int m, int r) {
    char temp[100][100];
    int i = l, j = m + 1, k = 0;

    while (i <= m && j <= r) {
        c++;
        if (strcmp(a[i], a[j]) <= 0)
            strcpy(temp[k++], a[i++]);
        else
            strcpy(temp[k++], a[j++]);
    }

    while (i <= m) strcpy(temp[k++], a[i++]);
    while (j <= r) strcpy(temp[k++], a[j++]);

    for (i = l, k = 0; i <= r; i++, k++)
        strcpy(a[i], temp[k]);
}

void mergeSort(char a[][100], int l, int r) {
    if (l >= r) return;

    int m = l + (r - l) / 2;
    mergeSort(a, l, m);
    mergeSort(a, m + 1, r);
    merge(a, l, m, r);
}

int main() {

19. Enter a String Array and Perform Quick Sort Through Iteration. Calculate Best and Worst Complexity with Time

#include <stdio.h>
#include <string.h>
#include <time.h>

int c = 0;

int partition(char a[][100], int low, int high) {
    char pivot[100], temp[100];
    strcpy(pivot, a[high]);
    int i = low - 1;

    for (int j = low; j < high; j++) {
        c++;
        if (strcmp(a[j], pivot) <= 0) {
            i++;
            strcpy(temp, a[i]);
            strcpy(a[i], a[j]);
            strcpy(a[j], temp);
        }
    }

    strcpy(temp, a[i + 1]);
    strcpy(a[i + 1], a[high]);
    strcpy(a[high], temp);

    return i + 1;
}

int main() {
    char a[100][100], temp[100];
    int stack[200], top = -1, n;
    clock_t start, end;

    printf("Enter number of strings: ");
    scanf("%d", &n);
    if (n <= 0 || n > 100) return 0;

    start = clock();

    stack[++top] = 0;
    stack[++top] = n - 1;

    while (top >= 0) {
        int high = stack[top--];
        int low = stack[top--];

        if (low < high) {
            int p = partition(a, low, high);

                stack[++top] = low;
                stack[++top] = p - 1;

            if (p + 1 < high) {
                stack[++top] = p + 1;
                stack[++top] = high;
            }
        }
    }

    end = clock();

    printf("Sorted strings:\n");
    for (int i = 0; i < n; i++) printf("%s\n", a[i]);
