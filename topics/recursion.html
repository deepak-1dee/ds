Recursion

1. Calculate factorial of a number using recursion.

#include <stdio.h>
long long factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}
int main() {
    int n;
    printf("Enter a non-negative integer (0-20): ");
    scanf("%d", &n);
    if (n < 0 || n > 20) printf("Enter a number from 0 to 20.\n");
    else printf("Factorial = %lld\n", factorial(n));
    return 0;
}

2. Calculate the Fibonacci number at a given position using recursion (position 0 starts with 0).

#include <stdio.h>
long long fibonacci(int n) {
    if (n == 0) return 0;
    if (n == 1) return 1;
    return fibonacci(n - 1) + fibonacci(n - 2);
}
int main() {
    int n;
    printf("Enter position (0-40): ");
    scanf("%d", &n);
    if (n < 0 || n > 40) printf("Enter a position from 0 to 40.\n");
    else printf("Fibonacci number = %lld\n", fibonacci(n));
    return 0;
}

3. Calculate the sum of natural numbers from 1 to 10 using recursion.

#include <stdio.h>
int sumNatural(int n) {
    if (n == 0) return 0;
    return n + sumNatural(n - 1);
}
int main() {
    printf("Sum of natural numbers from 1 to 10 = %d\n", sumNatural(10));
    return 0;
}

4. Convert a non-negative decimal number into binary using recursion.

#include <stdio.h>
void printBinary(unsigned int n) {
    if (n > 1) printBinary(n / 2);
    printf("%u", n % 2);
}
int main() {
    unsigned int n;
    printf("Enter a non-negative number: ");
    scanf("%u", &n);
    printf("Binary = ");
    printBinary(n);
    printf("\n");
    return 0;
}

5. Find the largest element in an integer array using recursion.

#include <stdio.h>
int largest(int a[], int n) {
    int rest;
    if (n == 1) return a[0];
    rest = largest(a, n - 1);
    return (a[n - 1] > rest) ? a[n - 1] : rest;
}
int main() {
    int a[100], n, i;
    printf("Enter number of elements (1-100): ");
    scanf("%d", &n);
    if (n < 1 || n > 100) { printf("Invalid size.\n"); return 0; }
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    printf("Largest element = %d\n", largest(a, n));
    return 0;
}

7. Enter a number and add all its digits using recursion.

#include <stdio.h>
int sumDigits(unsigned long long n) {
    if (n == 0) return 0;
    return (int)(n % 10) + sumDigits(n / 10);
}
int main() {
    long long n;
    printf("Enter an integer: ");
    scanf("%lld", &n);
    if (n < 0) n = -n;
    printf("Sum of digits = %d\n", sumDigits((unsigned long long)n));
    return 0;
}

8. Find the arithmetic mean of an integer array using recursion.

#include <stdio.h>
long long sumArray(int a[], int n) {
    if (n == 0) return 0;
    return a[n - 1] + sumArray(a, n - 1);
}
int main() {
    int a[100], n, i;
    long long sum;
    printf("Enter number of elements (1-100): ");
    scanf("%d", &n);
    if (n < 1 || n > 100) { printf("Invalid size.\n"); return 0; }
    printf("Enter %d elements: ", n);
    for (i = 0; i < n; i++) scanf("%d", &a[i]);
    sum = sumArray(a, n);
    printf("Arithmetic mean = %.2f\n", (double)sum / n);
    return 0;
}
